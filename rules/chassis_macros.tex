% ==============================================================================
% CHASSIS TEMPLATE MACROS FOR ROBOT BATTLES
% ==============================================================================
% Canonical macros ensuring identical chassis outlines, grid occupancy,
% wheel affectations, facing arrows, and Blue/Red/Ghost color palettes
% across all figures in the rulebook and print-and-play materials.
%
% Compass Heading Convention:
%   0 deg   = North (+Y)
%   45 deg  = North-East (CW)
%   90 deg  = East (+X)
%   135 deg = South-East (CW)
%   180 deg = South (-Y)
%   270 deg / -90 deg = West (-X)
%
% Usage:
%   \chassisSquare[options]{x}{y}{heading_deg}{grid_step}
%   \chassisTriangle[options]{x}{y}{heading_deg}{grid_step}
%   \chassisWide[options]{x}{y}{heading_deg}{grid_step}
%
% Options:
%   blue (default) | red | ghost
%   weapon = none (default) | bar | disc | forks
%   arrow = true (default) | false
%   center = true (default) | false
% ==============================================================================

\def\rbTeamBlue{blue}
\def\rbTeamRed{red}
\def\rbTeamGhost{ghost}
\def\rbWeaponBar{bar}
\def\rbWeaponDisc{disc}
\def\rbWeaponForks{forks}
\def\rbTrue{true}
\def\rbFalse{false}

\pgfkeys{
  /rbchassis/.is family, /rbchassis,
  team/.store in = \rbTeam,
  weapon/.store in = \rbWeapon,
  arrow/.store in = \rbArrow,
  center/.store in = \rbCenter,
  faded/.store in = \rbFaded,
  blue/.style = {team = blue},
  red/.style = {team = red},
  ghost/.style = {team = ghost},
}

\newcommand{\rbApplyStyle}{%
  \ifx\rbTeam\rbTeamGhost
    \def\rbBodyFill{gray!15}%
    \def\rbBodyDraw{gray!60}%
    \def\rbFadedFill{gray!10}%
    \def\rbFadedDraw{gray!35}%
    \def\rbWheelFill{gray!45}%
    \def\rbWheelDraw{gray!65}%
    \def\rbArrowDraw{gray!60}%
    \def\rbLineWidth{0.9pt}%
    \def\rbLineType{dashed}%
  \else
    \ifx\rbTeam\rbTeamRed
      \def\rbBodyFill{arenaRed!20}%
      \def\rbBodyDraw{arenaRed}%
      \def\rbFadedFill{arenaRed!15}%
      \def\rbFadedDraw{arenaRed!50}%
      \def\rbWheelFill{primaryDark}%
      \def\rbWheelDraw{primaryDark!80!black}%
      \def\rbArrowDraw{green!60!black}%
      \def\rbLineWidth{1.3pt}%
      \def\rbLineType{solid}%
    \else
      \def\rbBodyFill{accentBlue!25}%
      \def\rbBodyDraw{accentBlue}%
      \def\rbFadedFill{accentBlue!15}%
      \def\rbFadedDraw{accentBlue!50}%
      \def\rbWheelFill{primaryDark}%
      \def\rbWheelDraw{primaryDark!80!black}%
      \def\rbArrowDraw{green!60!black}%
      \def\rbLineWidth{1.3pt}%
      \def\rbLineType{solid}%
    \fi
  \fi
}

% ------------------------------------------------------------------------------
% 1. SQUARE CHASSIS (2x2 Occupancy)
% ------------------------------------------------------------------------------
\newcommand{\chassisSquare}[5][blue]{%
  \begingroup
  \pgfkeys{/rbchassis, team=blue, weapon=none, arrow=true, center=true, #1}%
  \rbApplyStyle
  \begin{scope}[shift={(#2,#3)}, rotate={-(#4)}, scale={(#5)}]
    % Drive wheels (4 wheels: 2 left, 2 right)
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] (-0.99, -0.68) rectangle (-0.85, -0.22);
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] (-0.99,  0.22) rectangle (-0.85,  0.68);
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] ( 0.85, -0.68) rectangle ( 0.99, -0.22);
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] ( 0.85,  0.22) rectangle ( 0.99,  0.68);
    % Main chassis body (2x2 occupancy, slightly inset)
    \draw[fill=\rbBodyFill, draw=\rbBodyDraw, line width=\rbLineWidth, \rbLineType, rounded corners=2pt]
      (-0.92, -0.92) rectangle (0.92, 0.92);
    % Center pin dot
    \ifx\rbCenter\rbTrue
      \fill[primaryDark] (0, 0) circle [radius=0.08];
    \fi
    % Weapon overlay
    \ifx\rbWeapon\rbWeaponBar
      \draw[line width=2.4pt, draw=arenaRed] (-0.75, 0.92) -- (0.75, 0.92);
    \fi
    \ifx\rbWeapon\rbWeaponDisc
      \fill[arenaRed] (0, 0.92) circle [radius=0.14];
    \fi
    % Facing arrow
    \ifx\rbArrow\rbTrue
      \draw[-{Triangle[width=2.4mm,length=3.2mm]}, line width=1.4pt, draw=\rbArrowDraw]
        (0, -0.35) -- (0, 0.65);
    \fi
  \end{scope}%
  \endgroup
}

% ------------------------------------------------------------------------------
% 3. TRIANGLE CHASSIS (Wedge Outline + Faded T-Tetris Occupancy)
% ------------------------------------------------------------------------------
\newcommand{\chassisTriangle}[5][blue]{%
  \begingroup
  \pgfkeys{/rbchassis, team=blue, weapon=none, arrow=true, center=true, faded=true, #1}%
  \rbApplyStyle
  \begin{scope}[shift={(#2,#3)}, rotate={-(#4)}, scale={(#5)}]
    % FADED T-TETRIS OCCUPANCY (Behind the wedge: 3x1 rear base + 1x1 front stem)
    \ifx\rbFaded\rbTrue
      \fill[\rbFadedFill] (-1.5, -0.5) rectangle (1.5, 0.5);
      \fill[\rbFadedFill] (-0.5,  0.5) rectangle (0.5, 1.5);
      % Subtle boundary & cell dividers
      \draw[\rbFadedDraw, dashed, line width=0.4pt]
        (-1.5, -0.5) -- (1.5, -0.5) -- (1.5, 0.5) -- (0.5, 0.5) --
        (0.5, 1.5) -- (-0.5, 1.5) -- (-0.5, 0.5) -- (-1.5, 0.5) -- cycle;
      \draw[\rbFadedDraw, dashed, line width=0.3pt] (-0.5, -0.5) -- (-0.5, 0.5);
      \draw[\rbFadedDraw, dashed, line width=0.3pt] ( 0.5, -0.5) -- ( 0.5, 0.5);
      \draw[\rbFadedDraw, dashed, line width=0.3pt] (-0.5,  0.5) -- ( 0.5, 0.5);
    \fi

    % Drive wheels on rear vertical side rails (axle through y=0)
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] (-1.47, -0.30) rectangle (-1.33, 0.30);
    \fill[\rbWheelFill, draw=\rbWheelDraw, line width=0.4pt] ( 1.33, -0.30) rectangle ( 1.47, 0.30);

    % WEDGE OUTLINE (rear flat at y=-0.42, side rails to y=0.40, apex at y=1.45)
    \draw[fill=\rbBodyFill, draw=\rbBodyDraw, line width=\rbLineWidth, \rbLineType, line join=round]
      (-1.40, -0.42) -- (1.40, -0.42) -- (1.40, 0.40) -- (0.0, 1.45) -- (-1.40, 0.40) -- cycle;

    % Center pin dot (at center of middle back tile)
    \ifx\rbCenter\rbTrue
      \fill[primaryDark] (0, 0) circle [radius=0.08];
    \fi

    % Weapon overlay
    \ifx\rbWeapon\rbWeaponBar
      \draw[line width=2.4pt, draw=arenaRed] (-0.45, 1.08) -- (0.0, 1.45) -- (0.45, 1.08);
    \fi
    \ifx\rbWeapon\rbWeaponDisc
      \fill[arenaRed] (0, 1.45) circle [radius=0.14];
    \fi

    % Facing arrow
    \ifx\rbArrow\rbTrue
      \draw[-{Triangle[width=2.4mm,length=3.2mm]}, line width=1.4pt, draw=\rbArrowDraw]
        (0, -0.05) -- (0, 1.15);
    \fi
  \end{scope}%
  \endgroup
}

% ==============================================================================
% 4. GRID MANAGEMENT & SNAP PLACEMENT MACROS
% ==============================================================================
% Enables defining arena grids with an origin (X0, Y0) and cell step size.
% All robot placement, contact dots, and grid vectors snap to integer/half-integer
% coordinates on the active grid.
%
% Usage:
%   \setGrid{x0}{y0}{step}
%   \gridPt{gx}{gy} -> TikZ coordinate ({x0 + gx*step}, {y0 + gy*step})
%   \gridDraw[style]{gx_min}{gy_min}{gx_max}{gy_max}
%   \gridSquare[options]{gx}{gy}{heading}
%   \gridTriangle[options]{gx}{gy}{heading}
%   \gridWide[options]{gx}{gy}{heading}
%   \gridDot[fill options]{gx}{gy}
%   \gridNode[options]{gx}{gy}{content}
% ==============================================================================

\def\rbGridXzero{0}
\def\rbGridYzero{0}
\def\rbGridStep{1}

\newcommand{\setGrid}[3]{%
  \def\rbGridXzero{#1}%
  \def\rbGridYzero{#2}%
  \def\rbGridStep{#3}%
}

\newcommand{\gridPt}[2]{({\rbGridXzero + (#1)*\rbGridStep}, {\rbGridYzero + (#2)*\rbGridStep})}

\newcommand{\gridDraw}[5][draw=boxBorder!60, line width=0.35pt]{%
  \draw[step=\rbGridStep, #1] \gridPt{#2}{#3} grid \gridPt{#4}{#5};%
}

\newcommand{\gridSquare}[4][blue]{%
  \pgfmathsetmacro{\rbCalcX}{\rbGridXzero + (#2)*\rbGridStep}%
  \pgfmathsetmacro{\rbCalcY}{\rbGridYzero + (#3)*\rbGridStep}%
  \chassisSquare[#1]{\rbCalcX}{\rbCalcY}{#4}{\rbGridStep}%
}

\newcommand{\gridWide}[4][blue]{%
  \pgfmathsetmacro{\rbCalcX}{\rbGridXzero + (#2)*\rbGridStep}%
  \pgfmathsetmacro{\rbCalcY}{\rbGridYzero + (#3)*\rbGridStep}%
  \chassisWide[#1]{\rbCalcX}{\rbCalcY}{#4}{\rbGridStep}%
}

\newcommand{\gridTriangle}[4][blue]{%
  \pgfmathsetmacro{\rbCalcX}{\rbGridXzero + (#2)*\rbGridStep}%
  \pgfmathsetmacro{\rbCalcY}{\rbGridYzero + (#3)*\rbGridStep}%
  \chassisTriangle[#1]{\rbCalcX}{\rbCalcY}{#4}{\rbGridStep}%
}

\newcommand{\gridDot}[3][arenaRed]{%
  \fill[#1] \gridPt{#2}{#3} circle (2.8pt);%
}

\newcommand{\gridNode}[4][]{%
  \node[#1] at \gridPt{#2}{#3} {#4};%
}
